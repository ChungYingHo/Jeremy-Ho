## 堆疊 (Stack)

### Java 中的堆疊 class

```java
import java.util.Stack;

public class stackInJava {
    public static void main(String[] args) {
        Stack<Integer> stack = new Stack<>();

        stack.push(1);
        stack.push(3);
        stack.push(2);
        stack.push(5);

        int top = stack.peek();
        System.out.println("The top is: " + top);

        int removeVal = stack.pop();
        System.out.println("Remove val is: " + removeVal);
        top = stack.peek();
        System.out.println("The top is: " + top);

        System.out.println("The stack size is: " + stack.size());
        System.out.println("Is the stack empty? Ans: " + stack.isEmpty());
        System.out.println("Is the stack empty? Ans: " + stack.empty());
    }
}
```

### 用 Array 實做 Stack
```java
public class Main {
    public static void main(String[] args) {
        System.out.println("=== Array-based Stack ===");
        StackArray stackA = new StackArray(5);
        stackA.push(10);
        stackA.push(20);
        stackA.push(30);
        System.out.println("Top element: " + stackA.peek());
        System.out.println("Pop: " + stackA.pop());
        System.out.println("Pop: " + stackA.pop());
        System.out.println("Is empty? " + stackA.isEmpty());
    }
}

class StackArray {
    private int[] arr;
    private int top; // 目前 stack 最上層的 index
    private int capacity; // array 的容量

    public StackArray(int size) {
        arr = new int[size];
        capacity = size;
        top = -1; // 初始時 stack 為空，故 top 設為 -1
    }

    public void push(int x) {
        if (top == capacity - 1) {
            System.out.println("Stack overflow!");
            return;
        }
        arr[++top] = x; // 先將 top 加 1，再將 x 放入該位置
    }

    public int pop() {
        if (isEmpty()) {
            System.out.println("Stack underflow!");
            return -1;
        }
        return arr[top--]; // 先回傳 top 位置的值，再將 top 減 1
    }

    public int peek() {
        if (isEmpty()) {
            System.out.println("Stack is empty!");
            return -1;
        }
        return arr[top];
    }

    public boolean isEmpty() {
        return top == -1;
    }
}
```

### 用 Linked List 實做 Stack
```java
public class Main {
    public static void main(String[] args) {
        System.out.println("\n=== LinkedList-based Stack ===");
        ListStack stackB = new ListStack();
        stackB.push("apple");
        stackB.push("banana");
        stackB.push("cherry");
        System.out.println("Top element: " + stackB.peek());
        System.out.println("Pop: " + stackB.pop());
        System.out.println("Pop: " + stackB.pop());
        System.out.println("Is empty? " + stackB.isEmpty());
    }
}

class ListStack {
    private Node top;

    private static class Node {
        String value;
        Node next;
        Node(String value) {
            this.value = value;
        }
    }

    public void push(String value) {
        Node newNode = new Node(value);
        newNode.next = top; // 新節點指向目前的 top
        top = newNode; // 更新 top 為新節點
    }

    public String pop() {
        if (isEmpty()) {
            System.out.println("Stack underflow!");
            return null;
        }
        String value = top.value;
        top = top.next;
        return value;
    }

    public String peek() {
        if (isEmpty()) {
            System.out.println("Stack is empty!");
            return null;
        }
        return top.value;
    }

    public boolean isEmpty() {
        return top == null;
    }
}
```

## 佇列 (Queue)
### Java 中的佇列 class

```java
import java.util.LinkedList;
import java.util.Queue;

public class queueInJava {
    public static void main(String[] args) {
        Queue<Integer> queue = new LinkedList<>();

        queue.add(1);
        queue.add(3);
        queue.add(2);
        queue.add(5);

        int front = queue.peek();
        System.out.println("The front is: " + front);

        int removeVal = queue.remove();
        System.out.println("Removed value: " + removeVal);

        front = queue.peek();
        System.out.println("The front is: " + front);

        System.out.println("The queue size is: " + queue.size());
        System.out.println("Is the queue empty? Ans: " + queue.isEmpty());
    }
}
```

### 用 Array 實做 Queue
```java
public class Main {
    public static void main(String[] args) {
        System.out.println("=== Array-based Queue ===");
        ArrayQueue queueA = new ArrayQueue(5);
        queueA.enqueue(10);
        queueA.enqueue(20);
        queueA.enqueue(30);
        System.out.println("Front element: " + queueA.peek());
        System.out.println("Dequeue: " + queueA.dequeue());
        System.out.println("Dequeue: " + queueA.dequeue());
        System.out.println("Is empty? " + queueA.isEmpty());
    }
}

class ArrayQueue {
    private int[] arr;
    private int front, rear, size, capacity;

    public ArrayQueue(int capacity) {
        this.capacity = capacity;
        arr = new int[capacity];
        front = 0;
        rear = -1;
        size = 0;
    }

    public void enqueue(int x) {
        if (isFull()) {
            System.out.println("Queue overflow!");
            return;
        }
        rear = (rear + 1) % capacity; // 環狀佇列
        arr[rear] = x;
        size++;
    }

    public int dequeue() {
        if (isEmpty()) {
            System.out.println("Queue underflow!");
            return -1;
        }
        int value = arr[front];
        front = (front + 1) % capacity;
        size--;
        return value;
    }

    public int peek() {
        if (isEmpty()) {
            System.out.println("Queue is empty!");
            return -1;
        }
        return arr[front];
    }

    public boolean isEmpty() {
        return size == 0;
    }

    public boolean isFull() {
        return size == capacity;
    }
}
```

### 用 Linked List 實做 Queue
```java
public class Main {
    public static void main(String[] args) {
        System.out.println("\n=== LinkedList-based Queue ===");
        ListQueue queueB = new ListQueue();
        queueB.enqueue("apple");
        queueB.enqueue("banana");
        queueB.enqueue("cherry");
        System.out.println("Front element: " + queueB.peek());
        System.out.println("Dequeue: " + queueB.dequeue());
        System.out.println("Dequeue: " + queueB.dequeue());
        System.out.println("Is empty? " + queueB.isEmpty());
    }
}

class ListQueue {
    private Node front, rear;

    private static class Node {
        String value;
        Node next;
        Node(String value) {
            this.value = value;
        }
    }

    public void enqueue(String value) {
        Node newNode = new Node(value);
        if (rear == null) { // queue 為空
            front = rear = newNode;
            return;
        }
        rear.next = newNode;
        rear = newNode;
    }

    public String dequeue() {
        if (isEmpty()) {
            System.out.println("Queue underflow!");
            return null;
        }
        String value = front.value;
        front = front.next;

        if (front == null) { // 若取完變空
            rear = null;
        }
        return value;
    }

    public String peek() {
        if (isEmpty()) {
            System.out.println("Queue is empty!");
            return null;
        }
        return front.value;
    }

    public boolean isEmpty() {
        return front == null;
    }
}
```
